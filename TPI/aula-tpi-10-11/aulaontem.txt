    //CRIAR ATRIBUTOS
    
    private String nome;
    private String endereco;
    private int idade;
    private double peso;

    //ENCAPSULAMENTO
    public String getNome() {
        return nome;
    }

    public void setNome(String nome) {
        this.nome = nome;
    }

    public String getEndereco() {
        return endereco;
    }

    public void setEndereco(String endereco) {
        this.endereco = endereco;
    }

    public int getIdade() {
        return idade;
    }

    public void setIdade(int idade) {
        this.idade = idade;
    }

    public double getPeso() {
        return peso;
    }

    public void setPeso(double peso) {
        this.peso = peso;
    }
    //MÉTODOS
    public void mostrarSystem(){
        System.out.println("Nome: " + getNome() +
                           "\n Endereço: " + getEndereco()+
                           "\n Idade: " + getIdade() +
                           "\n Peso: " + getPeso());
    }
    
    public void mostrarJOption(){
        JOptionPane.showMessageDialog(null,"Nome: " + getNome() +
                           "\n Endereço: " + getEndereco()+
                           "\n Idade: " + getIdade() +
                           "\n Peso: " + getPeso());
    }
    
    public void cadastrarSystem(){
        Scanner leia = new Scanner(System.in);
        System.out.println("Digite nome:");
        setNome(leia.next());
        System.out.println("Digite idade: ");
        setIdade(leia.nextInt());
        System.out.println("Digite o peso");
        setPeso(leia.nextDouble());
        System.out.println("Digite o endereço:");
        setEndereco(leia.next());
    }
    
    public void cadastrarJOption(){
        setNome(JOptionPane.showInputDialog("Digite nome: "));
        setIdade(Integer.parseInt(JOptionPane.showInputDialog("Idade:")));
        setPeso(Double.parseDouble(JOptionPane.showInputDialog("Peso: ")));
        setEndereco(JOptionPane.showInputDialog("Digite o endereço:"));
        
    }
